# Proxy de MangaDex como servicio independiente (el mismo manejador que usa Vite en desarrollo).
FROM node:22-alpine
WORKDIR /app
RUN echo '{"type":"module"}' > package.json
COPY vite/mangadexProxy.js ./vite/mangadexProxy.js
COPY deploy/mdx-server.mjs ./deploy/mdx-server.mjs
EXPOSE 3000
USER node
CMD ["node", "deploy/mdx-server.mjs"]
